from flask import Blueprint, render_template, request, redirect, url_for, flash, jsonify from flask_login import login_required, current_user from datetime import datetime from extensions import db from models import Venta, VentaDetalle, Cliente, Producto, Inventario, MovimientoInventario, Bitacora ventas_bp = Blueprint("ventas", __name__, template_folder="../templates/ventas") @ventas_bp.route("/") @login_required def index(): fecha_inicio = request.args.get("fecha_inicio", "") fecha_fin = request.args.get("fecha_fin", "") canal = request.args.get("canal", "") query = Venta.query if fecha_inicio: query = query.filter(Venta.fecha >= datetime.strptime(fecha_inicio, "%Y-%m-%d")) if fecha_fin: # Incluir todo el día final f_fin_dt = datetime.strptime(fecha_fin, "%Y-%m-%d").replace(hour=23, minute=59, second=59) query = query.filter(Venta.fecha <= f_fin_dt) if canal: query = query.filter(Venta.canal == canal) ventas = query.order_by(Venta.fecha.desc()).all() return render_template("ventas/index.html", ventas=ventas, fecha_inicio=fecha_inicio, fecha_fin=fecha_fin, canal=canal) @ventas_bp.route("/nueva", methods=["GET", "POST"]) @login_required def nueva(): clientes = Cliente.query.order_by(Cliente.nombre).all() # Productos con stock en ubicación 1 productos_con_stock = db.session.query(Producto, Inventario.existencia).join(Inventario, Inventario.id_producto == Producto.id).filter(Inventario.id_ubicacion == 1, Producto.activo == True).all() if request.method == "POST": id_cliente = request.form.get("id_cliente") canal = request.form.get("canal", "VENTA_DIRECTA") # Datos del carrito (enviados por campos ocultos o JSON desde el front) productos_ids = request.form.getlist("producto_id[]") cantidades = request.form.getlist("cantidad[]") precios = request.form.getlist("precio[]") if not productos_ids: flash("Debe agregar al menos un producto a la venta.", "danger") return redirect(url_for("ventas.nueva")) try: # Generar folio único basado en timestamp folio = f"V-{datetime.utcnow().strftime('%Y%m%d%H%M%S')}" nueva_venta = Venta( folio=folio, id_cliente=id_cliente if id_cliente else None, canal=canal, id_usuario=current_user.id, estatus="PAGADA", subtotal=0, total=0 ) db.session.add(nueva_venta) db.session.flush() subtotal_general = 0 for p_id, cant, prc in zip(productos_ids, cantidades, precios): cantidad = int(cant) precio_unitario = float(prc) producto = db.session.get(Producto, int(p_id)) if not producto: continue # Verificar inventario en ubicacion 1 inv = Inventario.query.filter_by(id_producto=producto.id, id_ubicacion=1).first() if not inv or inv.existencia < cantidad: raise Exception(f"Stock insuficiente para el producto: {producto.nombre}") # Descontar stock inv.existencia -= cantidad # Registrar movimiento de inventario mov = MovimientoInventario( id_producto=producto.id, id_ubicacion=1, tipo="VENTA", cantidad=-cantidad, referencia=folio, id_usuario=current_user.id ) db.session.add(mov) total_linea = cantidad * precio_unitario subtotal_general += total_linea detalle = VentaDetalle( id_venta=nueva_venta.id, id_producto=producto.id, cantidad=cantidad, precio_unitario=precio_unitario, costo_unitario=producto.precio_costo ) db.session.add(detalle) nueva_venta.subtotal = subtotal_general nueva_venta.total = subtotal_general # Bitácora bit = Bitacora( id_usuario=current_user.id, accion="CREAR_VENTA", entidad="ventas", entidad_id=nueva_venta.id, detalle=f"Folio {folio} creado por canal {canal}" ) db.session.add(bit) db.session.commit() flash(f"Venta {folio registrada exitosamente.", "success") return redirect(url_for("ventas.detalle", id=nueva_venta.id)) except Exception as e: db.session.rollback() flash(f"Error al procesar la venta: {str(e)}", "danger") return redirect(url_for("ventas.nueva")) return render_template("ventas/nueva.html", clientes=clientes, productos=productos_con_stock) @ventas_bp.route("/cliente-rapido", methods=["POST"]) @login_required def cliente_rapido(): """Endpoint AJAX para registrar un cliente al vuelo""" try: nombre = request.form.get("nombre") telefono = request.form.get("telefono", "") email = request.form.get("email", "") notas = request.form.get("notas", "") if not nombre: return jsonify({"success": False, "message": "El nombre es obligatorio."}), 400 nuevo_cliente = Cliente(nombre=nombre, telefono=telefono, email=email, notas=notas) db.session.add(nuevo_cliente) db.session.commit() return jsonify({ "success": True, "cliente": { "id": nuevo_cliente.id, "nombre": nuevo_cliente.nombre, "telefono": nuevo_cliente.telefono } }) except Exception as e: db.session.rollback() return jsonify({"success": False, "message": str(e)}), 500 @ventas_bp.route("//detalle") @login_required def detalle(id): venta = db.get_or_404(Venta, id) return render_template("ventas/detalle.html", venta=venta) @ventas_bp.route("//editar", methods=["GET", "POST"]) @login_required def editar(id): venta = db.get_or_404(Venta, id) clientes = Cliente.query.order_by(Cliente.nombre).all() if request.method == "POST": venta.id_cliente = request.form.get("id_cliente") or None venta.canal = request.form.get("canal", venta.canal) venta.estatus = request.form.get("estatus", venta.estatus) try: db.session.commit() flash("Ticket actualizado correctamente.", "success") return redirect(url_for("ventas.detalle", id=venta.id)) except Exception as e: db.session.rollback() flash(f"Error al actualizar: {str(e)}", "danger") return render_template("ventas/editar.html", venta=venta, clientes=clientes)
